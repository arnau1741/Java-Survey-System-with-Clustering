# 1. Ver la versión de Git
git --version

# 2. Ver el estado del repositorio (qué archivos cambiaron, no están añadidos, etc.)
git status

# 3. Agregar archivos al “staging” (prepararlos para commit)
git add ruta/al/archivo
# o para todos los cambios:
git add .

# 4. Hacer commit con mensaje
git commit -m "Mensaje descriptivo del cambio"

# 5. Subir los commits al repositorio remoto
git push

# 6. Traer cambios del remoto y fusionarlos con tu copia local
git pull

# 7. Ver historial de commits
git log

# 8. Revertir o deshacer commits

# Opción segura: revertir el último commit creando uno nuevo que lo deshace
git revert HEAD
git push

# Opción más agresiva: eliminar el último commit (cuidado)
git reset --hard HEAD~1
git push --force

# 9. Navegar dentro del proyecto
cd ruta/al/directorio

# 10. Verificar conexión SSH (cuando usás SSH)
ssh -T git@repo.fib.upc.es
